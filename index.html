DURUŞ ========== let poseActiveFlag=false, poseStream, poseVideo, poseCanvas, poseObj, poseInterval; async function startPose(){ stopAllModules('pose'); poseActiveFlag=true; poseVideo = document.getElementById('poseVideo'); poseCanvas = document.getElementById('poseCanvas'); poseStream = await navigator.mediaDevices.getUserMedia({ video: true }); poseVideo.srcObject = poseStream; await poseVideo.play(); poseObj = new Pose({ locateFile: f=>`https://cdn.jsdelivr.net/npm/@mediapipe/pose/${f}` }); poseObj.setOptions({ modelComplexity:1, smoothLandmarks:true }); poseObj.onResults((res)=>{ if(!poseActiveFlag) return; const ctx = poseCanvas.getContext('2d'); poseCanvas.width=poseVideo.videoWidth; poseCanvas.height=poseVideo.videoHeight; ctx.drawImage(poseVideo,0,0,poseCanvas.width,poseCanvas.height); let warn = ''; if(res.poseLandmarks){ for(let lm of res.poseLandmarks){ ctx.fillStyle='#0ff'; ctx.beginPath(); ctx.arc(lm.x*poseCanvas.width, lm.y*poseCanvas.height, 5,0,2*Math.PI); ctx.fill(); } const shoulderY = res.poseLandmarks[11]?.y, hipY = res.poseLandmarks[23]?.y; if(shoulderY && hipY && shoulderY+0.03 < hipY) warn='✅ Duruş yaxşı'; else warn='⚠️ Əyilmə ola bilər'; } else warn='❌ Skelet tapılmadı'; document.getElementById('poseStatsPanel').innerHTML = warn; }); poseInterval = setInterval(()=>{ if(poseActiveFlag && poseObj) poseObj.send({image:poseVideo}); }, 100); } function stopPose(){ if(poseInterval) clearInterval(poseInterval); if(poseStream) poseStream.getTracks().forEach(t=>t.stop()); if(poseVideo) poseVideo.srcObject=null; poseActiveFlag=false; document.getElementById('poseStatsPanel').innerHTML='Duruş analizi dayandırıldı'; const ctx = document.getElementById('poseCanvas').getContext('2d'); if(ctx) ctx.clearRect(0,0, ctx.canvas.width, ctx.canvas.height); } // ========= 4. EMOSIYA (face-api) ========== let emotionActive=false, emoStream, emoVideo, emoCanvas, emoInterval; async function loadFaceApiModels(){ await faceapi.nets.tinyFaceDetector.loadFromUri('https://cdn.jsdelivr.net/gh/justadudewhohacks/face-api.js@0.22.2/weights'); await faceapi.nets.faceExpressionNet.loadFromUri('https://cdn.jsdelivr.net/gh/justadudewhohacks/face-api.js@0.22.2/weights'); } async function startEmotion(){ stopAllModules('emotion'); if(typeof faceapi === 'undefined'){ alert('Face-api yüklənir, bir neçə saniyə gözlə'); return; } await loadFaceApiModels(); emotionActive=true; emoVideo = document.getElementById('emoVideo'); emoCanvas = document.getElementById('emoCanvas'); emoStream = await navigator.mediaDevices.getUserMedia({ video: true }); emoVideo.srcObject = emoStream; await emoVideo.play(); emoInterval = setInterval(async()=>{ if(!emotionActive) return; const detections = await faceapi.detectAllFaces(emoVideo, new faceapi.TinyFaceDetectorOptions()).withFaceExpressions(); const ctx = emoCanvas.getContext('2d'); emoCanvas.width = emoVideo.videoWidth; emoCanvas.height = emoVideo.videoHeight; ctx.drawImage(emoVideo,0,0, emoCanvas.width, emoCanvas.height); if(detections.length>0){ let exp = detections[0].expressions; let mainExpr = Object.keys(exp).reduce((a,b)=> exp[a]>exp[b] ? a : b); document.getElementById('emoStats').innerHTML = `😀 Emosiya: ${mainExpr} (${Math.round(exp[mainExpr]*100)}%)`; new faceapi.draw.DrawBox(detections[0].detection.box, { label: mainExpr }).draw(ctx); } else { document.getElementById('emoStats').innerHTML = '😐 Üz aşkarlanmadı'; } }, 200); } function stopEmotion(){ if(emoInterval) clearInterval(emoInterval); if(emoStream) emoStream.getTracks().forEach(t=>t.stop()); if(emoVideo) emoVideo.srcObject=null; emotionActive=false; document.getElementById('emoStats').innerHTML = 'Emosiya dayandırıldı'; const ctx = document.getElementById('emoCanvas').getContext('2d'); if(ctx) ctx.clearRect(0,0, ctx.canvas.width, ctx.canvas.height); } // ========= 5. DUYĞU BÜTÜNLƏMƏ SİMULYASİYASI ========== function refreshSensory(){ const light = Math.random()>0.7 ? 'parlaq' : (Math.random()<0.3 ? 'qaranlıq' : 'normal'); const sound = Math.random()>0.7 ? 'səs-küylü' : (Math.random()<0.3 ? 'çox sakit' : 'sakit'); const vib = Math.random()>0.8 ? 'hiss olunur' : 'yox'; document.getElementById('sensoryStats').innerHTML = `🔆 İşıq: ${light}   🔊 Səs: ${sound}   📳 Vibrasiya: ${vib}`; let advice = ''; if(light==='parlaq') advice += 'İşığı azaldın. '; if(sound==='səs-küylü') advice += 'Sakit otağa keçin. '; if(vib==='hiss olunur') advice += 'Titrişi aradan qaldırın. '; if(!advice) advice = 'Mühit rahat görünür, davam edin.'; document.getElementById('sensoryAdvice').innerHTML = `💡 Tövsiyə: ${advice}`; } // ========= 6. AUTISM – HANDS TRACKING ========== let handsActive=false, handsStream, handsVideo, handsCanvas, handsObj, handsInterval; async function startHands(){ stopAllModules('hands'); handsActive=true; handsVideo = document.getElementById('handsVideo'); handsCanvas = document.getElementById('handsCanvas'); handsStream = await navigator.mediaDevices.getUserMedia({ video: true }); handsVideo.srcObject = handsStream; await handsVideo.play(); handsObj = new Hands({ locateFile: f=>`https://cdn.jsdelivr.net/npm/@mediapipe/hands/${f}` }); handsObj.setOptions({ maxNumHands:2, minDetectionConfidence:0.5 }); handsObj.onResults((res)=>{ if(!handsActive) return; const ctx = handsCanvas.getContext('2d'); handsCanvas.width = handsVideo.videoWidth; handsCanvas.height = handsVideo.videoHeight; ctx.drawImage(handsVideo,0,0,handsCanvas.width,handsCanvas.height); let msg = 'Əl aşkarlanmadı.'; if(res.multiHandLandmarks && res.multiHandLandmarks.length>0){ msg = `👐 ${res.multiHandLandmarks.length} əl göründü. Təkrarlanan hərəkətə diqqət!`; for(let landmarks of res.multiHandLandmarks){ for(let lm of landmarks){ ctx.fillStyle='#ff0'; ctx.beginPath(); ctx.arc(lm.x*handsCanvas.width, lm.y*handsCanvas.height, 4,0,2*Math.PI); ctx.fill(); } } } document.getElementById('handsStats').innerHTML = msg; }); handsInterval = setInterval(()=>{ if(handsActive && handsObj) handsObj.send({image: handsVideo}); }, 100); } function stopHands(){ if(handsInterval) clearInterval(handsInterval); if(handsStream) handsStream.getTracks().forEach(t=>t.stop()); if(handsVideo) handsVideo.srcObject=null; handsActive=false; document.getElementById('handsStats').innerHTML = 'Əl izləmə dayandırıldı.'; const ctx = document.getElementById('handsCanvas').getContext('2d'); if(ctx) ctx.clearRect(0,0, ctx.canvas.width, ctx.canvas.height); } // ========= 7. PLAN GENERATOR ========== function generatePlan(){ const age = parseInt(document.getElementById('childAge').value); const domain = document.getElementById('childDomain').value; let plan = ''; if(domain === 'İnkişaf geriliyi') plan = `🎯 ${age} yaş üçün: incə motorika oyunları, nitq stimulyasiyası, gündəlik 40 dəq fərdi məşğələ.`; else if(domain === 'Autizm') plan = `🧩 ${age} yaş üçün: vizual dəstək, strukturlaşdırılmış oyun, sosial hekayələr, stereo hərəkətləri azaltmaq üçün alternativ fəaliyyətlər.`; else if(domain === 'Daun sindromu') plan = `🌟 ${age} yaş üçün: artikulyasiya gimnastikası, ümumi motor inkişaf, oxuya hazırlıq, musiqi terapiyası.`; else plan = `🗣️ ${age} yaş nitq gecikməsi: səs oyunları, nəfəs məşqləri, şəkil kartları ilə dialoq, həftədə 3 dəfə loqoped.`; document.getElementById('planOutput').innerHTML = `📋 ${plan}`; } // ----- DÜYMƏLERİN BİRİKDİRİLMƏSİ ----- window.onload = async ()=>{ document.getElementById('btnAttStart').onclick = startAttention; document.getElementById('btnAttStop').onclick = stopAttention; document.getElementById('btnSpeechStart').onclick = startSpeech; document.getElementById('btnSpeechStop').onclick = stopSpeech; document.getElementById('btnPoseStart').onclick = startPose; document.getElementById('btnPoseStop').onclick = stopPose; document.getElementById('btnEmoStart').onclick = startEmotion; document.getElementById('btnEmoStop').onclick = stopEmotion; document.getElementById('btnSensoryRefresh').onclick = refreshSensory; document.getElementById('btnHandsStart').onclick = startHands; document.getElementById('btnHandsStop').onclick = stopHands; document.getElementById('btnGenPlan').onclick = generatePlan; refreshSensory(); // ilkin // face-api modelləri əvvəlcədən yüklə if(typeof faceapi !== 'undefined') await loadFaceApiModels(); };
